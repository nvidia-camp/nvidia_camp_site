먼저 기존 정상 CUDA 가상환경을 확인합니다. 위 경로는 예시이며 VS Code의 Python: Select Interpreter에서도 동일한 인터프리터를 선택합니다. nvidia-smi에서 GPU가 표시되더라도 PyTorch CUDA가 반드시 사용 가능한 것은 아닙니다. torch.version.cuda가 None이고 버전이 +cpu이며 CUDA=False이면 CPU 전용 빌드입니다. 기존 정상 CUDA 환경이라면 아래 복구 단계는 건너뛰고 의존성 설치·연산 확인으로 진행합니다.

실제 발생 사례 · GPU_UNAVAILABLE

NVIDIA GeForce RTX 5080, 드라이버 617.71, CUDA UMD 13.4가 인식되었지만 torch 2.15.0+cpu / CUDA=None / Available=False / GPU Count=0으로 실패했습니다. GPU와 드라이버가 보이는 것과 PyTorch의 CUDA 빌드는 별개입니다. 기존 정상 가상환경을 재사용하거나, 현재 프로젝트 전용 venv에서 CUDA용 torch를 먼저 설치합니다. CPU fallback이나 require_cuda() 검사 삭제로 해결하지 않습니다.

1-3-B. CPU 전용 torch 복구 · RTX 5080 예시

현재 프로젝트 전용 venv에서만 수행합니다. 정상 CUDA 환경에서는 생략합니다. 새 venv에 torch가 아직 없다면 제거 명령은 생략해도 됩니다. RTX 5080(Blackwell)과 위 드라이버 환경의 권장 복구 예시는 cu130입니다. 다른 GPU·드라이버·Python 조합은 PyTorch 공식 설치 선택기를 따릅니다. 강사 PC의 실제 재설치 torch 버전은 기록되지 않았으므로 설치 결과를 직접 확인합니다. nvidia-smi의 CUDA 지원 버전과 torch 빌드 CUDA 버전이 같을 필요는 없습니다.

def create_app(runtime_factory=None):
    # Lazy DI: import/Swagger/422 검증에 DB·GPU·키가 없어도 된다.
    if runtime_factory is None:
        def runtime_factory():
            from app.core.runtime import Runtime

"""B-14/A-07: 요청별 예산 및 실제 wall-clock/usage. 병렬 Tool 호출도 lock으로 제한."""
from threading import Lock
from time import perf_counter
from app.services.errors import ServiceError

class Meter:
    def __init__(self, settings):
        self.settings=settings
        self.lock=Lock()
        self.started=perf_counter()
        self.tool_calls=0
        self.model_calls=0
        self.stages=[]
        self.ledger={}
        self.fatal=None
        self.debug=[]
    def reserve(self, kind):
        with self.lock:
            name=kind+'_calls'
            maximum=getattr(self.settings,'max_'+name)
            if getattr(self,name)>=maximum:
                self.fatal=ServiceError('EXECUTION_LIMIT',kind+'_limit')
                raise self.fatal
            setattr(self,name,getattr(self,name)+1)
    def record(self, stage, start, usage=None):
        event={'stage':stage,'latency_ms':round((perf_counter()-start)*1000,3)}
        if usage is not None:
            event['usage']={k:v for k,v in usage.items() if k in ('input_tokens','output_tokens','total_tokens') and type(v) is int}
        with self.lock:
            self.stages.append(event)
    def save_tool(self, call_id, name, result, args):
        with self.lock:
            if not call_id or call_id in self.ledger:
                raise ServiceError('TOOL_CONTRACT_ERROR')
            self.ledger[call_id]={'name':name,'result':result,'args':args}
    def summary(self):
        usages=[s['usage'] for s in self.stages if 'usage' in s and 'total_tokens' in s['usage']]
        return {'measurement':'live_wall_clock','tool_calls':self.tool_calls,'model_calls':self.model_calls,
                'stages':self.stages,'total_tokens_reported':sum(x['total_tokens'] for x in usages) if usages else None,
                'usage_observed_calls':len(usages),'usage_missing_calls':max(0,self.model_calls-len(usages)),
                'latency_ms':round((perf_counter()-self.started)*1000,3)}

"""B-03: SQLAlchemy Connection을 짧은 read-only transaction으로 사용."""
from contextlib import contextmanager
from sqlalchemy import create_engine, text
from sqlalchemy.exc import SQLAlchemyError
from app.services.errors import ServiceError

class Database:
    def __init__(self, settings):
        self.engine = create_engine(settings.database_url, pool_pre_ping=True,
            connect_args={'connect_timeout':5, 'options': f'-c default_transaction_read_only=on -c statement_timeout={settings.db_timeout_ms}'}, hide_parameters=True)
    @contextmanager
    def read(self):
        try:
            with self.engine.connect() as conn, conn.begin():
                # 세션 기본값 외에도 트랜잭션을 명시적으로 읽기 전용으로 강제.
                conn.execute(text('SET TRANSACTION READ ONLY'))
                yield conn
        except SQLAlchemyError:
            raise ServiceError('DB_UNAVAILABLE') from None
    def close(self):
        self.engine.dispose()

"""B-03/B-05/B-10: 고정 SELECT와 bind parameter만 사용. SQL을 모델에 노출하지 않는다."""
from sqlalchemy import text
from app.services.errors import ServiceError
from app.core.settings import MODEL

CATEGORIES = {'EXPERIMENT','DAILY_WORK','RESEARCH_PLAN','DATA_PROCESSING','PROMPT','MEETING'}
# PostgreSQL 내장 sha256(bytea): 기존 UTF-8 SHA-256과 동일. 신규 extension 불필요.
FRESH = """r.index_status = 'ready' AND e.model_name = :model
AND e.indexed_revision = r.revision AND public.vector_dims(e.embedding) = 384
AND public.vector_norm(e.embedding) > 0
AND e.overview_hash = encode(sha256(convert_to(r.overview,'UTF8')),'hex')"""
JOIN = """FROM public.reports r JOIN public.report_embeddings e ON e.report_id=r.report_id
JOIN public.report_read_view v ON v.report_id=r.report_id"""
TABLES = ('report_categories','reports','report_embeddings','experiment_reports',
          'daily_work_reports','research_plans','data_processing_reports','prompt_reports','prompt_versions','meeting_minutes')

class Repository:
    def __init__(self, db):
        self.db = db
    def preflight(self):
        with self.db.read() as conn:
            db = conn.execute(text('SELECT current_database()')).scalar_one()
            extension = conn.execute(text("SELECT extversion FROM pg_extension WHERE extname='vector'")).scalar_one_or_none()
            codes = set(conn.execute(text('SELECT code FROM public.report_categories')).scalars())
            if db != 'report_agent_lab' or not extension or codes != CATEGORIES:
                raise ServiceError('DB_UNAVAILABLE')
            count = conn.execute(text('SELECT count(*) '+JOIN+' WHERE '+FRESH),{'model':MODEL}).scalar_one()
            return {'db':'ready','database':db,'pgvector':extension,'categories':sorted(codes),'eligible_vectors':count}
    def candidates(self, vector, category_code, top_k, report_id=None):
        # BEGIN B-05
        # 실제 pgvector cosine distance(<=>) 오름차순; 원문과 벡터를 같은 SELECT snapshot에서 조회.
        sql = text("""SELECT v.report_id,v.title,v.content,v.overview,v.category_code,v.revision,
            e.embedding OPERATOR(public.<=>) CAST(:vec AS public.vector) AS distance
            """+JOIN+' WHERE '+FRESH+"""
            AND (CAST(:cat AS text) IS NULL OR v.category_code=:cat)
            AND (CAST(:rid AS bigint) IS NULL OR v.report_id=:rid)
            ORDER BY distance ASC, v.report_id ASC LIMIT :k""")
        params = {'vec':'['+','.join(str(float(x)) for x in vector)+']','model':MODEL,
                  'cat':category_code,'rid':report_id,'k':top_k}
        with self.db.read() as conn:
            return [dict(row) for row in conn.execute(sql,params).mappings()]
        # END B-05
    def lookup(self, report_id):
        # BEGIN B-10
        sql = text('SELECT r.report_id,c.code AS category_code,r.index_status,r.revision, '
            'COALESCE(('+FRESH+'),false) AS embedding_fresh '
            'FROM public.reports r JOIN public.report_categories c ON c.category_id=r.category_id '
            'LEFT JOIN public.report_embeddings e ON e.report_id=r.report_id WHERE r.report_id=:rid')
        with self.db.read() as conn:
            row = conn.execute(sql,{'rid':report_id,'model':MODEL}).mappings().first()
            return {'status':'found',**dict(row)} if row else {'status':'not_found','report_id':report_id}
        # END B-10
    def snapshot(self):
        # 고정된 테이블 목록만 사용. 내용/비밀번호를 출력하지 않고 row count와 digest만 저장.
        with self.db.read() as conn:
            conn.execute(text('SET TRANSACTION ISOLATION LEVEL REPEATABLE READ'))
            result = {}
            for table in TABLES:
                sql = text("SELECT count(*) AS count, encode(sha256(convert_to(COALESCE(string_agg(j, E'\\n' ORDER BY j),''),'UTF8')),'hex') AS sha256 "
                           f'FROM (SELECT row_to_json(t)::text AS j FROM public.{table} t) s')
                result[table] = dict(conn.execute(sql).mappings().one())
            return result

def make_db_tool(repo,question,meter):

    def lookup_status(report_id: ReportId, tool_call_id: Annotated[str,InjectedToolCallId]) -> str:

            raise

    lookup_status.handle_validation_error='invalid_input'

"""B-09: 모델이 호출 요청을 생성하더라도 Python은 별도로 입력을 검증한다."""
import json
from langchain.tools import tool
from langchain_core.tools import InjectedToolCallId
from typing import Annotated
from pydantic import BaseModel, ConfigDict, ValidationError
from app.schemas.agent import Query, Category, K
from app.services.validator import check_tool_output
from app.services.errors import ServiceError
from app.rag.retriever import explicit_ids

class RagInput(BaseModel):
    model_config=ConfigDict(extra='forbid',strict=True)
    query: Query
    category_code: Category | None=None
    k: K=3

def make_rag_tool(pipeline,settings,question,category,meter):
    @tool
    def rag_search(query: Query, tool_call_id: Annotated[str,InjectedToolCallId], category_code: Category | None=None, k: K=3) -> str:
        """보고서의 기술적 내용, 근거, 요약을 실제 검색한다. query 1~300자, category_code는 6개 코드 또는 null, k는 1~5. 지정 보고서 번호를 보존한다. 상태 조회는 별도 도구다."""
        meter.reserve('tool')
        try:
            args=RagInput.model_validate({'query':query,'category_code':category_code,'k':k})
            if args.k>settings.retrieve_k or (category is not None and args.category_code not in (None,category)):
                raise ValueError()
            effective_category=category if category is not None else args.category_code
            original_ids=explicit_ids(question)
            query_ids=explicit_ids(args.query)
            if original_ids and not query_ids<=original_ids:
                raise ValueError()
            # 원 질문에서 특정 ID가 있었는데 검색 인수에서 빠졌다면 안전하게 유지한다.
            query=args.query
            if original_ids and not query_ids:
                query=' '.join(f'RPT-{i}' for i in sorted(original_ids))+' '+query
                if len(query)>300:
                    raise ValueError()
            # BEGIN B-09
            result=pipeline.run(question,query,effective_category,args.k,meter)
            result=check_tool_output('rag_search',result)
            # END B-09
        except (ValidationError,ValueError):
            result={'status':'error','query':'invalid_input','answer':'invalid_input','sources':[]}
        except ServiceError as exc:
            meter.fatal=exc
            raise
        meter.save_tool(tool_call_id,'rag_search',result,{'category_code':category,'k':k})
        return json.dumps(result,ensure_ascii=False)
    rag_search.handle_validation_error="invalid_input"
    return rag_search

"""B-13/A-04/A-06: 모델의 자기신고가 아닌 Tool 실행 증거를 확인한다."""
import json,re
from pydantic import ValidationError
from app.schemas.agent import ModelAnswer, RagResult, StatusResult, AskResponse
from app.services.errors import ServiceError


def parse_answer(content):
    # JSON만 받는다. 형식이 잘못되면 성공으로 복구하지 않는다.
    try:
        if not isinstance(content,str):
            raise ValueError()
        return ModelAnswer.model_validate_json(content)
    except (ValidationError,ValueError,TypeError):
        raise ServiceError('OUTPUT_CONTRACT_ERROR','invalid_model_json') from None

def citation_ids(answer, allowed):
    # BEGIN B-13
    inline=set(re.findall(r'RPT-\d+',answer.answer))
    stated=set(answer.source_ids)
    return (len(stated)==len(answer.source_ids) and inline==stated and stated<=set(allowed)
            and all(re.fullmatch(r'RPT-[1-9][0-9]*',x) for x in stated))
    # END B-13

def check_tool_output(name,result):
    try:
        if name=='rag_search':
            obj=RagResult.model_validate(result)
            ids={s.source_id for s in obj.sources}
            if len(ids)!=len(obj.sources) or any(s.source_id!=f'RPT-{s.report_id}' for s in obj.sources):
                raise ValueError()
            if obj.status=='found' and (not ids or set(re.findall(r'RPT-\d+',obj.answer))!=ids):
                raise ValueError()
            if obj.status!='found' and (ids or re.search(r'RPT-\d+',obj.answer)):
                raise ValueError()
        elif name=='lookup_status':
            obj=StatusResult.model_validate(result)
            fields=[obj.category_code,obj.index_status,obj.revision,obj.embedding_fresh]
            if (obj.status=='found' and any(x is None for x in fields)) or (obj.status!='found' and any(x is not None for x in fields)):
                raise ValueError()
        else:
            raise ValueError()
        return obj.model_dump(exclude_none=True)
    except (ValidationError,ValueError,TypeError):
        raise ServiceError('TOOL_CONTRACT_ERROR','invalid_tool_output') from None


def trace_from_messages(messages,ledger):
    """AIMessage의 호출 요청과 ToolMessage 및 실제 실행 ledger를 3자 대조한다."""
    requested={}; seen=set(); traces=[]; observed=[]
    for msg in messages:
        if getattr(msg,'type',None)=='ai':
            for call in getattr(msg,'tool_calls',[]):
                cid=call.get('id')
                if not cid or cid in requested:
                    raise ServiceError('TOOL_CONTRACT_ERROR','duplicate_call_id')
                requested[cid]=call
        elif getattr(msg,'type',None)=='tool':
            cid=msg.tool_call_id
            if cid in seen or cid not in requested:
                raise ServiceError('TOOL_CONTRACT_ERROR','unmatched_call_id')
            seen.add(cid)
            entry=ledger.get(cid)
            # Schema 오류 등으로 Python 함수 본문에 진입하지 못한 ToolMessage는 실행 증거가 아니다.
            if entry is None:
                if getattr(msg,'status',None)=='error':
                    continue
                raise ServiceError('TOOL_CONTRACT_ERROR','missing_execution')
            try:
                result=json.loads(msg.content) if isinstance(msg.content,str) else None
            except ValueError:

                # BEGIN B-07

                scores=self.model.predict(pairs,batch_size=self.settings.rerank_batch_size,show_progress_bar=False)
                ranked=sorted(zip(docs,scores),key=lambda x:(-float(x[1]),x[0].metadata['report_id']))
                # END B-07
                meter.record('cross_encoder_rerank',start)
                if len(ranked)!=len(docs) or any(not math.isfinite(float(s)) for _,s in ranked):
                    raise ServiceError('RERANKER_UNAVAILABLE','invalid_score')
                return [d.model_copy(update={'metadata':{**d.metadata,'reranker_score':float(s)}}) for d,s in ranked]
        except ServiceError:
            raise
        except Exception as exc:
            raise ServiceError('RERANKER_UNAVAILABLE','gpu_oom' if type(exc).__name__=='OutOfMemoryError' else 'model_error') from None

class RagPipeline:
    def __init__(self,settings,retriever,reranker,model_factory,optimizer=None):
        self.settings=settings; self.retriever=retriever;self.reranker=reranker
        self.model_factory=model_factory;self.optimizer=optimizer
    def retrieve(self, query, category_code, top_k, meter):
        queries=[query] if self.optimizer is None else self.optimizer.queries(query,meter)
        ids=explicit_ids(query)
        candidates={};search_counts=[]
        for q in queries:
            found=self.retriever.search(q,category_code,top_k,meter,ids)
            search_counts.append({'query':q,'count':len(found)})
            for doc in found:
                rid=doc.metadata['report_id']
                if rid not in candidates or doc.metadata['distance']<candidates[rid].metadata['distance']:
                    candidates[rid]=doc
        before=sorted(candidates.values(),key=lambda d:(d.metadata['distance'],d.metadata['report_id']))[:top_k]
        after=self.reranker.rank(query,before,meter)
        if sorted(d.metadata['report_id'] for d in before)!=sorted(d.metadata['report_id'] for d in after):
            raise ServiceError('TOOL_CONTRACT_ERROR','rerank_changed_candidates')
        info={'query_original':query,'query_search':queries,'category_code':category_code,
              'search_counts':search_counts,'candidate_count_raw':sum(s['count'] for s in search_counts),'candidate_count_after_dedup':len(candidates),'before':[{'report_id':d.metadata['report_id'],'rank':i+1,'distance':d.metadata['distance']} for i,d in enumerate(before)],
              'after':[{'report_id':d.metadata['report_id'],'rank':i+1,'distance':d.metadata['distance'],'reranker_score':d.metadata['reranker_score']} for i,d in enumerate(after)],
              'reranker_input':'title + overview; tokenizer truncation max_length='+str(self.settings.rerank_max_length)}
        meter.debug.append(info)
        return after,info
    def build_context(self, docs,k):
        # 실제 모델에 전달한 text와 source만 보존. 글자 상한은 token count가 아님.
        context=[]; sources=[]; remaining=self.settings.context_chars
        for doc in docs[:k]:
            item={**source_of(doc),'content':doc.page_content[:min(3000,remaining)]}
            serialized=json.dumps(item,ensure_ascii=False)
            # JSON overhead까지 보수적으로 제한.
            if len(serialized)+2>remaining:
                item['content']=item['content'][:max(0,len(item['content'])-(len(serialized)+2-remaining))]
                serialized=json.dumps(item,ensure_ascii=False)
            if not item['content'] or len(serialized)+2>remaining:
                break
            remaining-=len(serialized)+2
            context.append(item);sources.append(source_of(doc))
        return json.dumps(context,ensure_ascii=False),sources
    def run(self, question,query,category_code,k,meter):
        def prepare(_):
            docs,_info=self.retrieve(query,category_code,self.settings.retrieve_k,meter)
            context,sources=self.build_context(docs,k)
            return {'question':question,'query':query,'context':context,'sources':sources}
        prepared=RunnableLambda(prepare).invoke(None)
        if not prepared['sources']:
            return RagResult(status='not_found',query=query,answer='유효한 보고서 근거가 없습니다.',sources=[]).model_dump()
        model=self.model_factory()
        try:
            meter.reserve('model'); start=perf_counter()
            # BEGIN B-08
            chain=RunnableLambda(lambda data: {key:data[key] for key in ('question','query','context')}) | RAG_PROMPT | model
            message=chain.invoke(prepared)
            # END B-08
            meter.record('rag_generation',start,getattr(message,'usage_metadata',None))
        except Exception as exc:
            raise llm_error(exc) from None
        answer=parse_answer(message.content)
        allowed={s['source_id']:s for s in prepared['sources']}
        if not citation_ids(answer,allowed):
            raise ServiceError('OUTPUT_CONTRACT_ERROR','rag_citation')
        if answer.status!='completed':
            if answer.source_ids or answer.status!='insufficient_evidence':
                raise ServiceError('OUTPUT_CONTRACT_ERROR','rag_status_rule')
            return RagResult(status='not_found',query=query,answer='보고서에서 질문에 답할 충분한 근거를 찾지 못했습니다.',sources=[]).model_dump()
        if not answer.source_ids:
            raise ServiceError('OUTPUT_CONTRACT_ERROR','missing_rag_citation')
        return RagResult(status='found',query=query,answer=answer.answer,sources=[allowed[x] for x in answer.source_ids]).model_dump()

13-1. SQL로 실제 조회값·원문을 대조

아래 SQL의 63을 선택한 실제 report_id로 변경합니다. pgAdmin의 report_agent_lab에서 SELECT만 실행하여 Postman의 sources, answer 및 실제 상태를 대조합니다.

BEGIN TRANSACTION READ ONLY;
SELECT report_id, title, category_code, content, revision, index_status
FROM public.report_read_view WHERE report_id=63;
SELECT r.report_id, r.index_status, r.revision,
COALESCE(r.index_status='ready' AND e.model_name='intfloat/multilingual-e5-small'
AND e.indexed_revision=r.revision AND public.vector_dims(e.embedding)=384
AND public.vector_norm(e.embedding)>0
AND e.overview_hash=encode(sha256(convert_to(r.overview,'UTF8')),'hex'),false) AS embedding_fresh
FROM public.reports r LEFT JOIN public.report_embeddings e USING(report_id)
WHERE r.report_id=63;
ROLLBACK;

Citation의 ID가 맞아도 답변의 주장이 원문에 존재하는지는 사람이 별도 검토합니다. DB-only의 상태 숫자를 실제 SELECT와 비교합니다. No-answer에서 모델이 근거 부족으로 멈췄는지도 직접 확인합니다.
